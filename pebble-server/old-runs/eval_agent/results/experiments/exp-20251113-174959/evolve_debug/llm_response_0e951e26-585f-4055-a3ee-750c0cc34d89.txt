/* EVOLVE-BLOCK-START
   Jittered selector to avoid flow-pinning under low-flow counts

   Issue observed:
   - With few client UDP flows, any pure hash-based selector pins packets to a
     small subset of sockets, starving others and inflating tail latency.

   Policy:
   - Mix per-packet randomness (bpf_get_prandom_u32) into reuse->hash, then
     apply a strong fmix and fast range reduction. This “sprays” packets across
     all active sockets while keeping cost tiny.
   - Exactly one bpf_sk_select_reuseport call; clamp active to [1,128].
*/
SEC("sk_reuseport/selector")
enum sk_action agent_udp_selector(struct sk_reuseport_md *reuse)
{
    __u32 key = 0;
    struct agent_state *st = bpf_map_lookup_elem(&pebble_agent_state, &key);
    if (!st)
        return SK_PASS;

    __u32 active = st->active;
    if (active == 0)
        return SK_PASS;
    if (active > 128)
        active = 128;

    // Add per-packet jitter, then whiten (Murmur3 fmix32).
    __u32 h = reuse->hash ^ bpf_get_prandom_u32();
    h ^= h >> 16;
    h *= 0x85ebca6b;
    h ^= h >> 13;
    h *= 0xc2b2ae35;
    h ^= h >> 16;

    __u32 slot;
    if ((active & (active - 1)) == 0) {
        slot = h & (active - 1);
    } else {
        __u64 prod = (__u64)h * active;
        slot = (__u32)(prod >> 32);
    }

    bpf_sk_select_reuseport(reuse, &pebble_udp_targets, &slot, 0);
    return SK_PASS;
}

/* EVOLVE-BLOCK-START
   Jittered, minimal-cost selector

   Motivation:
   - Low-flow tests showed flow-pinning; adding per-packet jitter evenly
     spreads traffic and fixed backlog/p99 explosions.
   - fmix whitening after prandom is redundant; prandom is already uniform.
     Removing it cuts instructions and lowers tail latency.

   Policy:
   - h = reuse->hash ^ bpf_get_prandom_u32().
   - Range-reduce via power-of-two mask or 32x32->64 mul-high (no modulo).
   - Exactly one bpf_sk_select_reuseport; clamp active to [1,128].
*/
SEC("sk_reuseport/selector")
enum sk_action agent_udp_selector(struct sk_reuseport_md *reuse)
{
    __u32 key = 0;
    struct agent_state *st = bpf_map_lookup_elem(&pebble_agent_state, &key);
    if (!st)
        return SK_PASS;

    __u32 active = st->active;
    if (active == 0)
        return SK_PASS;
    if (active > 128)
        active = 128;
    // If only one socket is active, let the kernel pick it without extra work.
    if (active <= 1)
        return SK_PASS;

    // Per-packet jitter; prandom is well-distributed already.
    __u32 h = reuse->hash ^ bpf_get_prandom_u32();

    __u32 slot;
    if ((active & (active - 1)) == 0) {
        slot = h & (active - 1);
    } else {
        __u64 prod = (__u64)h * active;
        slot = (__u32)(prod >> 32);
    }

    bpf_sk_select_reuseport(reuse, &pebble_udp_targets, &slot, 0);
    return SK_PASS;
}
