 Approach:

 - Fewer instructions than a 128-iteration unrolled probe loop.

SEC("sk_reuseport/selector")
enum sk_action agent_udp_selector(struct sk_reuseport_md *reuse)
{
    __u32 key = 0;
    struct agent_state *state = bpf_map_lookup_elem(&pebble_agent_state, &key);
    if (!state || state->active == 0)
        return SK_PASS;

    __u32 start = reuse->hash % state->active;

    #pragma clang loop unroll(full)
    for (int i = 0; i < 128; i++) {
        if (i >= state->active)
            break;

        __u32 slot = start + i;
        if (slot >= state->active)
            slot -= state->active;

        if (bpf_sk_select_reuseport(reuse, &pebble_udp_targets, &slot, 0) == 0)
            return SK_PASS;
    }

    return SK_PASS;
}

/* EVOLVE-BLOCK-START
   Agent UDP reuseport selector (single-shot, low-latency)

   Rationale:
   - The kernel honors only the first successful bpf_sk_select_reuseport call.
     Extra attempts are ignored and just burn cycles.
   - Derive the slot from reuse->hash with fast range reduction:
       * power-of-two active: bitmask
       * otherwise: 32x32->64 multiply and take high 32 bits (avoid modulo)
   - Clamp active to [1,128]; on any issue, return SK_PASS for kernel default.
*/
SEC("sk_reuseport/selector")
enum sk_action agent_udp_selector(struct sk_reuseport_md *reuse)
{
    __u32 key = 0;
    struct agent_state *st = bpf_map_lookup_elem(&pebble_agent_state, &key);
    if (!st)
        return SK_PASS;

    __u32 active = st->active;
    if (active == 0)
        return SK_PASS;
    if (active > 128)
        active = 128;

    __u32 slot;
    if ((active & (active - 1)) == 0) {
        // Power-of-two fast path
        slot = reuse->hash & (active - 1);
    } else {
        // General case: fast range reduction without modulo
        __u64 prod = (__u64)reuse->hash * active;
        slot = (__u32)(prod >> 32);
    }

    // Single selection attempt; fallback handled by returning SK_PASS.
    bpf_sk_select_reuseport(reuse, &pebble_udp_targets, &slot, 0);
    return SK_PASS;
}
